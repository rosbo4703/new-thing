<script>

    // Automatically set today's date on load

    document.getElementById('logDate').valueAsDate = new Date();

 

    function saveLogEntry() {

        const entryData = {

            date: document.getElementById('logDate').value,

            hoursLogged: document.getElementById('hoursLogged').value,

            currentSprintGoal: document.getElementById('sprintGoal').value,

            progressAndAccomplishments: document.getElementById('progress').value,

            technicalChallengeAndReasoning: document.getElementById('challenge').value,

            softSkillFocus: document.getElementById('softSkill').value,

            sourceCitations: document.getElementById('sources').value,

            timestamp: new Date().toISOString()

        };

 

        // Basic validation check

        if (!entryData.date || !entryData.hoursLogged || !entryData.currentSprintGoal) {

            alert("Please fill out the header fields (Date, Hours, Sprint Goal) before saving.");

            return;

        }

 

        // Convert object to a downloadable JSON file

        const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(entryData, null, 2));

        const downloadAnchor = document.createElement('a');

        downloadAnchor.setAttribute("href", dataStr);

       

        // Format filename using date and goal snippet

        const safeGoal = entryData.currentSprintGoal.replace(/[^a-z0-9]/gi, '_').toLowerCase();

        downloadAnchor.setAttribute("download", `sprint_log_${entryData.date}_${safeGoal.substring(0, 15)}.json`);

       

        document.body.appendChild(downloadAnchor);

        downloadAnchor.click();

        downloadAnchor.remove();

    }

</script>
